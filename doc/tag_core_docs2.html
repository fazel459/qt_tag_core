۱. نمای کلی پروژه

Tag Core یک هسته مرکزی SCADA/Historian است که داده‌های صنعتی را از درایورهای مختلف دریافت، پردازش، ذخیره و از طریق API های WebSocket و REST در اختیار GUI های مستقل قرار می‌دهد.

📥 Ingestion

دریافت داده از Modbus TCP/RTU، MQTT، Simulator و OPC UA

🧮 Processing

Scaling، Filtering، Rounding، Deadband و Exception Reporting

🚨 Rules & Alarms

۹ نوع آلارم با hysteresis، delay و severity

🗄 Historian

TimescaleDB با Continuous Aggregates، Compression و Retention

🔌 API Layer

WebSocket (real-time) + REST (CRUD/Query) + Command Channel

📊 Dashboards

داشبوردهای file-based با پشتیبانی QML/HTML/JSON

📈 Reports

گزارش CSV/JSON از تاریخچه، آلارم‌ها و خلاصه روزانه

۲. ساختار داده‌ها

TagValue — داده زنده در TagBus

struct TagValue {
    qint64  tagId = 0;
    QString tagName;
    QDateTime timestamp;
    double  rawValue = 0.0;
    double  engineeringValue = 0.0;
    Quality quality = Quality::Good;      // Good / Bad / Uncertain
    SourceKind source = SourceKind::Simulator; // RealDriver/Simulator/Calculated/Manual/Replay
    quint64 sequence = 0;
};

TagDefinition — تعریف تگ (جدول tags)

struct TagDefinition {
    qint64 tagId;            QString tagName;
    QString sourceType;      QString dataType;      QString engUnits;
    double rawMin, rawMax, engMin, engMax;
    QString scalingType;     double slope, offset;
    double deadband, storageDeadband, alarmHysteresis;
    int heartbeatIntervalMs;
    QString softwareFilter, softwareFilterConfig;
    QString simProfile;      qint64 driverId;
    QString addressConfig;   bool enabled, clampEnabled;
};

DriverDefinition — تعریف درایور (جدول drivers)

struct DriverDefinition {
    qint64 driverId;
    QString name;
    QString type;              // simulator / modbus_tcp / modbus_rtu / mqtt
    QString connectionConfig;  # JSON string
    int pollingIntervalMs;
    bool enabled;
};

ساختار Rule ها

struct ThresholdRule { ruleId, tagId, low, high, lowlow, highhigh,
    highHysteresis, lowHysteresis, lowLowHysteresis, highHighHysteresis,
    onDelayMs, offDelayMs, hasLow/hasHigh/hasLowLow/hasHighHigh };

struct RangeViolationRule { ruleId, tagId, minValue, maxValue, severity };
struct RateOfChangeRule  { ruleId, tagId, maxRatePerSecond, windowMs, severity };
struct StuckValueRule    { ruleId, tagId, stuckDurationMs, epsilon, severity };
struct BooleanRule       { ruleId, tagId, alarmOnTrue, alarmOnFalse, durationMs, severity };
struct NotificationRule  { notificationRuleId, name, severityFilter,
    alarmTypeFilter, channel, channelConfig, throttleMs, enabled };
struct ComputedTag       { computedTagId, tagId, expression, updateMode, updateIntervalMs, enabled };

struct DashboardDefinition {
