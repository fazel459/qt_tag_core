🌐 ۱۲. مرجع — WebSocket جدید

⚠️ محدودیت Postman
Postman پشتیبانی محدودی از WebSocket دارد. برای تست کامل، از مرورگر یا ابزارهایی مثل wscat یا کتابخانه‌های JS استفاده کنید. در زیر مثال‌های آماده آورده شده است.

URL‌های اتصال

URLعملکرد
ws://localhost:8080/wsاتصال ساده
ws://localhost:8080/ws/alarms/allاشتراک همه آلارم‌ها
ws://localhost:8080/ws/tags/batch?ids=1,2,3اشتراک تگ‌ها (batch)
ws://localhost:8080/ws/dashboard/main?ids=1,2اشتراک داشبورد

مثال JavaScript — اتصال و subscribe

const ws = new WebSocket('ws://localhost:8080/ws/tags/batch?ids=1,2,3');

ws.onopen = () => console.log('Connected');

ws.onmessage = (e) => {
  const msg = JSON.parse(e.data);
  switch (msg.type) {
    case 'hello':
      console.log('Session:', msg.session);
      break;
    case 'tags.batch':
      msg.updates.forEach(u => console.log(`Tag ${u.tag_id}: ${u.value} [${u.quality}]`));
      break;
    case 'tags.snapshot':
      console.log('Snapshot:', msg.tags);
      break;
    case 'alarm.event':
      console.log(`ALARM [${msg.severity}] ${msg.category}:`, msg);
      break;
    case 'alarm.ack':
      console.log('Alarm acked by:', msg.user_name);
      break;
    case 'response':
      console.log(msg.ok ? 'CMD OK:' : 'CMD ERR:', msg.data || msg.error);
      break;
    case 'error':
      console.error('WS Error:', msg.message);
      break;
  }
};

// Heartbeat: سرور هر 30s ping می‌فرستد، مرورگر خودکار pong می‌دهد

مثال JavaScript — ارسال Command

// نوشتن مقدار روی تگ
ws.send(JSON.stringify({
  type: 'command',
  id: 'cmd-' + Date.now(),
  op: 'write_tag',
  payload: { tag_id: 1, value: 75.5 }
}));

// پاسخ در msg.type === 'response' دریافت می‌شود

مثال JavaScript — Subscribe/Unsubscribe دستی

// Subscribe
ws.send(JSON.stringify({
  op: 'subscribe',
  channel: ['alarms/all', 'system/status'],
  tags: [1, 2, 3],
  dashboard: ['main'],
  delivery: 'batch'
}));

// Unsubscribe
ws.send(JSON.stringify({
  op: 'unsubscribe',
  tags: [1, 2]
}));

// تغییر حالت دریافت
ws.send(JSON.stringify({ op: 'options', delivery: 'event' }));

// Snapshot
ws.send(JSON.stringify({ op: 'get_snapshot', tags: [1, 2, 3] }));
