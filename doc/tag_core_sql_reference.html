🚀 راه‌اندازی اولیه دیتابیس

پیش‌نیازها: PostgreSQL 14+ و TimescaleDB Extension

1. ایجاد دیتابیس

CREATE DATABASE
-- اتصال به psql -- psql -U postgres -- ایجاد دیتابیس CREATE DATABASE tagsdb; -- اتصال به دیتابیس \c tagsdb; -- فعال‌سازی TimescaleDB CREATE EXTENSION IF NOT EXISTS timescaledb;

2. ایجاد کاربر (اختیاری)

CREATE USER
-- ایجاد کاربر جدید CREATE USER tagcore WITH PASSWORD 'your_password'; -- اعطای دسترسی‌ها GRANT ALL PRIVILEGES ON DATABASE tagsdb TO tagcore; GRANT ALL PRIVILEGES ON ALL TABLES IN SCHEMA public TO tagcore; GRANT ALL PRIVILEGES ON ALL SEQUENCES IN SCHEMA public TO tagcore;

3. بررسی صحت نصب

SELECT
-- بررسی نسخه TimescaleDB SELECT extversion FROM pg_extension WHERE extname = 'timescaledb'; -- بررسی PostgreSQL version SELECT version();

📋 ایجاد جداول

این جداول به صورت خودکار توسط DbManager ساخته می‌شوند، اما برای مرجع در اینجا آورده شده‌اند.

1. جدول tags

CREATE TABLE
CREATE TABLE IF NOT EXISTS tags ( tag_id BIGSERIAL PRIMARY KEY, tag_name TEXT UNIQUE NOT NULL, driver_id BIGINT REFERENCES drivers(driver_id), source_type TEXT DEFAULT 'real_driver', data_type TEXT DEFAULT 'float', eng_units TEXT, raw_min DOUBLE PRECISION DEFAULT 0, raw_max DOUBLE PRECISION DEFAULT 100, eng_min DOUBLE PRECISION DEFAULT 0, eng_max DOUBLE PRECISION DEFAULT 100, scaling_type TEXT DEFAULT 'linear', scaling_slope DOUBLE PRECISION DEFAULT 1.0, scaling_offset DOUBLE PRECISION DEFAULT 0.0, clamp_enabled BOOLEAN DEFAULT TRUE, deadband DOUBLE PRECISION DEFAULT 0.5, storage_deadband DOUBLE PRECISION DEFAULT 0.5, alarm_hysteresis DOUBLE PRECISION DEFAULT -1, heartbeat_interval_ms INT DEFAULT 30000, software_filter_type TEXT DEFAULT 'none', software_filter_config TEXT DEFAULT '{}', sim_profile TEXT DEFAULT 'sine', address_config TEXT DEFAULT '{}', enabled BOOLEAN DEFAULT TRUE, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() );

2. جدول tag_values_raw (Hypertable)

CREATE TABLE + HYPERTABLE
CREATE TABLE IF NOT EXISTS tag_values_raw ( time TIMESTAMPTZ NOT NULL, tag_id BIGINT NOT NULL, raw_value DOUBLE PRECISION, eng_value DOUBLE PRECISION, quality INT DEFAULT 0, source TEXT ); -- تبدیل به Hypertable SELECT create_hypertable('tag_values_raw', 'time', if_not_exists => TRUE); -- ایجاد ایندکس CREATE INDEX IF NOT EXISTS idx_tag_values_raw_tag_time ON tag_values_raw(tag_id, time DESC);

3. جدول tag_current_state

CREATE TABLE
CREATE TABLE IF NOT EXISTS tag_current_state ( tag_id BIGINT PRIMARY KEY, last_value DOUBLE PRECISION, last_quality INT, last_timestamp TIMESTAMPTZ, updated_at TIMESTAMPTZ DEFAULT now() );

4. جدول alarms

CREATE TABLE
CREATE TABLE IF NOT EXISTS alarms ( alarm_id BIGSERIAL PRIMARY KEY, tag_id BIGINT, rule_type TEXT, alarm_type TEXT, severity TEXT, priority TEXT, state TEXT, value DOUBLE PRECISION, threshold DOUBLE PRECISION, message TEXT, active_time TIMESTAMPTZ DEFAULT now(), clear_time TIMESTAMPTZ, ack_time TIMESTAMPTZ, ack_user TEXT ); CREATE INDEX IF NOT EXISTS idx_alarms_tag_state ON alarms(tag_id, state);

5. جدول alarm_events

CREATE TABLE
CREATE TABLE IF NOT EXISTS alarm_events ( event_id BIGSERIAL PRIMARY KEY, alarm_id BIGINT REFERENCES alarms(alarm_id), event_type TEXT NOT NULL, event_time TIMESTAMPTZ DEFAULT now(), event_data TEXT, user_name TEXT ); CREATE INDEX IF NOT EXISTS idx_alarm_events_alarm_id ON alarm_events(alarm_id);

6. جدول drivers

CREATE TABLE
CREATE TABLE IF NOT EXISTS drivers ( driver_id BIGSERIAL PRIMARY KEY, name TEXT UNIQUE NOT NULL, type TEXT NOT NULL, connection_config TEXT DEFAULT '{}', polling_interval_ms INT DEFAULT 1000, enabled BOOLEAN DEFAULT TRUE, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() );

7. جدول threshold_rules

CREATE TABLE
CREATE TABLE IF NOT EXISTS threshold_rules ( rule_id BIGSERIAL PRIMARY KEY, tag_id BIGINT NOT NULL REFERENCES tags(tag_id), low DOUBLE PRECISION, high DOUBLE PRECISION, lowlow DOUBLE PRECISION, highhigh DOUBLE PRECISION, high_hysteresis DOUBLE PRECISION, low_hysteresis DOUBLE PRECISION, lowlow_hysteresis DOUBLE PRECISION, highhigh_hysteresis DOUBLE PRECISION, on_delay_ms INT, off_delay_ms INT, enabled BOOLEAN DEFAULT TRUE, updated_at TIMESTAMPTZ DEFAULT now() );

8. جدول range_violation_rules

CREATE TABLE
CREATE TABLE IF NOT EXISTS range_violation_rules ( rule_id BIGSERIAL PRIMARY KEY, tag_id BIGINT NOT NULL REFERENCES tags(tag_id), min_value DOUBLE PRECISION, max_value DOUBLE PRECISION, severity TEXT DEFAULT 'high', enabled BOOLEAN DEFAULT TRUE, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() );

9. جدول rate_of_change_rules

CREATE TABLE
CREATE TABLE IF NOT EXISTS rate_of_change_rules ( rule_id BIGSERIAL PRIMARY KEY, tag_id BIGINT NOT NULL REFERENCES tags(tag_id), max_rate_per_second DOUBLE PRECISION NOT NULL, window_ms INT DEFAULT 5000, severity TEXT DEFAULT 'high', enabled BOOLEAN DEFAULT TRUE, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() );

10. جدول stuck_value_rules

CREATE TABLE
CREATE TABLE IF NOT EXISTS stuck_value_rules ( rule_id BIGSERIAL PRIMARY KEY, tag_id BIGINT NOT NULL REFERENCES tags(tag_id), stuck_duration_ms INT DEFAULT 60000, epsilon DOUBLE PRECISION DEFAULT 0.01, severity TEXT DEFAULT 'medium', enabled BOOLEAN DEFAULT TRUE, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() );

11. جدول boolean_rules

CREATE TABLE
CREATE TABLE IF NOT EXISTS boolean_rules ( rule_id BIGSERIAL PRIMARY KEY, tag_id BIGINT NOT NULL REFERENCES tags(tag_id), alarm_on_true BOOLEAN DEFAULT FALSE, alarm_on_false BOOLEAN DEFAULT FALSE, duration_ms INT DEFAULT 1000, severity TEXT DEFAULT 'medium', enabled BOOLEAN DEFAULT TRUE, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() );

12. جدول computed_tags

CREATE TABLE
CREATE TABLE IF NOT EXISTS computed_tags ( computed_tag_id BIGSERIAL PRIMARY KEY, tag_id BIGINT NOT NULL REFERENCES tags(tag_id), expression TEXT NOT NULL, update_mode TEXT DEFAULT 'on_change', update_interval_ms INT DEFAULT 1000, enabled BOOLEAN DEFAULT TRUE, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() ); CREATE INDEX IF NOT EXISTS idx_computed_tags_tag ON computed_tags(tag_id);

13. جدول notification_rules

CREATE TABLE
CREATE TABLE IF NOT EXISTS notification_rules ( notification_rule_id BIGSERIAL PRIMARY KEY, name TEXT NOT NULL, severity_filter TEXT, alarm_type_filter TEXT, channel TEXT NOT NULL, channel_config TEXT DEFAULT '{}', throttle_ms INT DEFAULT 60000, enabled BOOLEAN DEFAULT TRUE, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now() );

14. جدول notification_log

CREATE TABLE
CREATE TABLE IF NOT EXISTS notification_log ( notification_id BIGSERIAL PRIMARY KEY, alarm_id BIGINT, notification_rule_id BIGINT, channel TEXT, status TEXT, message TEXT, sent_at TIMESTAMPTZ DEFAULT now() ); CREATE INDEX IF NOT EXISTS idx_notification_log_alarm ON notification_log(alarm_id);

15. جدول archive_log

CREATE TABLE
CREATE TABLE IF NOT EXISTS archive_log ( archive_id BIGSERIAL PRIMARY KEY, table_name TEXT NOT NULL, file_path TEXT NOT NULL, start_time TIMESTAMPTZ, end_time TIMESTAMPTZ, record_count BIGINT DEFAULT 0, file_size_bytes BIGINT DEFAULT 0, status TEXT DEFAULT 'completed', created_at TIMESTAMPTZ DEFAULT now() );

16. جدول system_settings

CREATE TABLE
CREATE TABLE IF NOT EXISTS system_settings ( key TEXT PRIMARY KEY, value_text TEXT, value_int INT, value_double DOUBLE PRECISION, value_bool BOOLEAN, description TEXT, updated_at TIMESTAMPTZ DEFAULT now() );

📊 Continuous Aggregates

1. tag_values_1min

CREATE MATERIALIZED VIEW
CREATE MATERIALIZED VIEW IF NOT EXISTS tag_values_1min WITH (timescaledb.continuous) AS SELECT time_bucket(INTERVAL '1 minute', time) AS bucket, tag_id, AVG(eng_value) AS avg_value, MIN(eng_value) AS min_value, MAX(eng_value) AS max_value, FIRST(eng_value, time) AS first_value, LAST(eng_value, time) AS last_value, COUNT(*) AS sample_count FROM tag_values_raw GROUP BY bucket, tag_id WITH NO DATA;

2. tag_values_5min

CREATE MATERIALIZED VIEW
CREATE MATERIALIZED VIEW IF NOT EXISTS tag_values_5min WITH (timescaledb.continuous) AS SELECT time_bucket(INTERVAL '5 minutes', time) AS bucket, tag_id, AVG(eng_value) AS avg_value, MIN(eng_value) AS min_value, MAX(eng_value) AS max_value, FIRST(eng_value, time) AS first_value, LAST(eng_value, time) AS last_value, COUNT(*) AS sample_count FROM tag_values_raw GROUP BY bucket, tag_id WITH NO DATA;

3. tag_values_1hour

CREATE MATERIALIZED VIEW
CREATE MATERIALIZED VIEW IF NOT EXISTS tag_values_1hour WITH (timescaledb.continuous) AS SELECT time_bucket(INTERVAL '1 hour', time) AS bucket, tag_id, AVG(eng_value) AS avg_value, MIN(eng_value) AS min_value, MAX(eng_value) AS max_value, FIRST(eng_value, time) AS first_value, LAST(eng_value, time) AS last_value, COUNT(*) AS sample_count FROM tag_values_raw GROUP BY bucket, tag_id WITH NO DATA;

4. tag_values_1day

CREATE MATERIALIZED VIEW
CREATE MATERIALIZED VIEW IF NOT EXISTS tag_values_1day WITH (timescaledb.continuous) AS SELECT time_bucket(INTERVAL '1 day', time) AS bucket, tag_id, AVG(eng_value) AS avg_value, MIN(eng_value) AS min_value, MAX(eng_value) AS max_value, FIRST(eng_value, time) AS first_value, LAST(eng_value, time) AS last_value, COUNT(*) AS sample_count FROM tag_values_raw GROUP BY bucket, tag_id WITH NO DATA;

⚙️ Policies (Compression, Retention, Refresh)

1. Continuous Aggregate Policies

ADD CONTINUOUS AGGREGATE POLICY
SELECT add_continuous_aggregate_policy('tag_values_1min', start_offset => INTERVAL '3 minutes', end_offset => INTERVAL '1 minute', schedule_interval => INTERVAL '1 minute', if_not_exists => TRUE ); SELECT add_continuous_aggregate_policy('tag_values_5min', start_offset => INTERVAL '15 minutes', end_offset => INTERVAL '5 minutes', schedule_interval => INTERVAL '5 minutes', if_not_exists => TRUE ); SELECT add_continuous_aggregate_policy('tag_values_1hour', start_offset => INTERVAL '3 hours', end_offset => INTERVAL '1 hour', schedule_interval => INTERVAL '1 hour', if_not_exists => TRUE ); SELECT add_continuous_aggregate_policy('tag_values_1day', start_offset => INTERVAL '3 days', end_offset => INTERVAL '1 day', schedule_interval => INTERVAL '1 day', if_not_exists => TRUE );

2. Compression Policies

ADD COMPRESSION POLICY
-- فعال‌سازی compression برای tag_values_raw ALTER TABLE tag_values_raw SET ( timescaledb.compress, timescaledb.compress_segmentby = 'tag_id', timescaledb.compress_orderby = 'time DESC' ); SELECT add_compression_policy('tag_values_raw', compress_after => INTERVAL '7 days', if_not_exists => TRUE ); -- فعال‌سازی compression برای tag_values_1min ALTER MATERIALIZED VIEW tag_values_1min SET ( timescaledb.compress, timescaledb.compress_segmentby = 'tag_id' ); SELECT add_compression_policy('tag_values_1min', compress_after => INTERVAL '30 days', if_not_exists => TRUE ); -- فعال‌سازی compression برای tag_values_5min ALTER MATERIALIZED VIEW tag_values_5min SET ( timescaledb.compress, timescaledb.compress_segmentby = 'tag_id' ); SELECT add_compression_policy('tag_values_5min', compress_after => INTERVAL '90 days', if_not_exists => TRUE ); -- فعال‌سازی compression برای tag_values_1hour ALTER MATERIALIZED VIEW tag_values_1hour SET ( timescaledb.compress, timescaledb.compress_segmentby = 'tag_id' ); SELECT add_compression_policy('tag_values_1hour', compress_after => INTERVAL '1 year', if_not_exists => TRUE );

3. Retention Policies

ADD RETENTION POLICY
SELECT add_retention_policy('tag_values_raw', drop_after => INTERVAL '30 days', if_not_exists => TRUE ); SELECT add_retention_policy('tag_values_1min', drop_after => INTERVAL '90 days', if_not_exists => TRUE ); SELECT add_retention_policy('tag_values_5min', drop_after => INTERVAL '1 year', if_not_exists => TRUE ); SELECT add_retention_policy('tag_values_1hour', drop_after => INTERVAL '5 years', if_not_exists => TRUE );

🔍 Query های کاربردی

1. آلارم‌های فعال

SELECT
SELECT alarm_id, tag_id, alarm_type, severity, state, value, threshold, message, active_time FROM alarms WHERE state = 'active' ORDER BY active_time DESC;

2. تاریخچه آلارم‌ها

SELECT
SELECT alarm_id, tag_id, alarm_type, severity, state, value, threshold, message, active_time, clear_time FROM alarms ORDER BY active_time DESC LIMIT 20;

3. رویدادهای آلارم

SELECT
SELECT e.event_id, e.alarm_id, e.event_type, e.event_time, a.alarm_type, a.severity FROM alarm_events e JOIN alarms a ON e.alarm_id = a.alarm_id ORDER BY e.event_time DESC LIMIT 20;

4. داده‌های خام یک تگ

SELECT
SELECT time, tag_id, raw_value, eng_value, quality, source FROM tag_values_raw WHERE tag_id = 1 ORDER BY time DESC LIMIT 20;
